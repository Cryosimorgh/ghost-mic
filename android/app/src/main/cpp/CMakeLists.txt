cmake_minimum_required(VERSION 3.22.1)
project(ghost_denoiser LANGUAGES C CXX)
set(RN rnnoise/src)
add_library(rnnoise STATIC ${RN}/denoise.c ${RN}/rnn.c ${RN}/rnn_data.c ${RN}/pitch.c ${RN}/kiss_fft.c ${RN}/celt_lpc.c)
set_target_properties(rnnoise PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 99)
target_include_directories(rnnoise PUBLIC rnnoise/include PRIVATE ${RN})
target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD HAVE_LRINTF)
target_compile_options(rnnoise PRIVATE -O3)
add_library(ghost_denoiser SHARED denoiser.cpp)
target_compile_features(ghost_denoiser PRIVATE cxx_std_17)
target_link_libraries(ghost_denoiser PRIVATE rnnoise m)
target_link_options(ghost_denoiser PRIVATE "-Wl,-z,max-page-size=16384")
